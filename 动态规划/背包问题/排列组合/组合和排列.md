# 组合个数问题

### 0-1背包: 每类物品最多只能选择一个

物品体积volumes, 背包容量capacity, 求恰好装满背包的不同组合数

```java
int combNum(int[] volumes, int capacity) {
    int n = volumes.length;
    // dp[i][j]: 只使用物品[0...i-1], 背包容量为j时, 恰好装满背包的不同组合数
    int[][] dp = new int[n + 1][capacity + 1];
    // 初始化
    for(int i = 0; i <= n; i++) {
        dp[i][0] = 1; // 只有一种方案能装满容量为0的背包, 即任何物品都不选
    }
    // 递推
    for(int i = 1; i <= n; i++) {
        for(int j = 0; j <= capacity; j++) {
            if(j < volumes[i - 1])
                dp[i][j] = dp[i - 1][j]; // 此时背包容量不足以选择物品i-1
            else
                dp[i][j] = dp[i - 1][j] + dp[i - 1][j - volumes[i - 1]]; // 此时背包容量足以选择物品i-1
        }
    }
    return dp[n][capacity]
}

// 空间优化1: dp[i]行只依赖于dp[i-1]行, 故可将空间降低一维
int combNum(int[] volumes, int capacity) {
    int n = volumes.length;
    // dp[i][j]: 只使用物品[0...i-1], 背包容量为j时, 恰好装满背包的不同组合数
    int[] dp = new int[capacity];
    // 初始化
    dp[0] = 1; 
    // 递推
    for(int i = 1; i <= n; i++) {
      	int[] temp = new int[capacity + 1]; // 临时数组
        for(int j = 0; j <= capacity; j++) {
            if(j < volumes[i - 1])
                temp[j] = dp[j]; 
            else
                temp[j] = dp[j] + dp[j - volumes[i - 1]];
        }
      	dp = temp;
    }
    return dp[capacity]
}

    dp[0] = 1;

        for(int j = capacity; j >= 0; j--) {

              	dp[j] = dp[j];
          	else
              	dp[j] = dp[j] + dp[j - nums[i - 1]];
        }
    }

}
```
